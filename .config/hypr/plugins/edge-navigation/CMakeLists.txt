cmake_minimum_required(VERSION 3.25)
project(edge_navigation VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_HYPRLAND_PLUGIN "Build the Hyprland 0.57.2 plugin; never load it" ON)
if(BUILD_HYPRLAND_PLUGIN)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland=0.57.2)
    pkg_check_modules(LUA REQUIRED IMPORTED_TARGET lua5.5)
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-server)
    add_library(edge_navigation MODULE src/Plugin.cpp)
    target_link_libraries(edge_navigation PRIVATE PkgConfig::HYPRLAND PkgConfig::LUA PkgConfig::WAYLAND)
    target_compile_options(edge_navigation PRIVATE -Wall -Wextra)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(edge_navigation PRIVATE -fno-gnu-unique)
    endif()
    target_compile_definitions(edge_navigation PRIVATE EDGE_NAVIGATION_VERSION="${PROJECT_VERSION}")
    set_target_properties(edge_navigation PROPERTIES PREFIX "" OUTPUT_NAME "edge-navigation"
        CXX_VISIBILITY_PRESET default VISIBILITY_INLINES_HIDDEN YES)
endif()
